    count++;
    console.log(count);
  }
}

fn();

// OUTPUT - 1 2 INNER FUNCTION CLOSES OVER COUNT - preserving its value due to closure.

---------------------------------------OBJECTS REF IN SETS------------------------------------------------------------
let obj = { a: 1, b: 2 };
let str = new Set();
str.add(obj);
obj = null;
console.log(str.has(obj));

	OUTPUT : false.  WHEN WE STORE AN OBJECT IN SET, IT STORES REFERENCE TO THE OBJECT, NOT OBJECT NAME or its content.. 
	as we are reassigning obj to null here. SET STILL HAS REFERENCE TO OBJ 
	
	DETAIL -- 
			obj → 0x123     , str → contains 0x123
			WHEN WE DO obj = null, we only change obj to null, wherease Set still has 0x123
			but when we do str.has(obj) - HERE OBJ is null. and null !=0x123 so it returns false. 
	
let obj = { a: 1, b: 2 };
let str = new Set();
str.add(obj);
let ref = obj;
obj = null;
console.log(str.has(ref)); // ✅ true
